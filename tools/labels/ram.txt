; ---- object tables (index X = object slot; slot 0 = player, enemies start at slot $0C)

0490 objLife        16   ; enemy life; "one hit kills" cheat patches the SBC that subtracts from it (array of 16 object slots; slot 0 = player; Data Crystal addresses are slot $0C)

05B0 objSpriteDown  16   ; (?) sprite id while lying down (array of 16 object slots; slot 0 = player; Data Crystal addresses are slot $0C)

; ---- misc
0200 oamBuffer 256              ; OAM DMA source ($4014 <- $02)
0600 palBuffer 32               ; palette buffer copied to $3F00 by NMI when palUpdateFlag is set

; ---- MMC3 / PPU shadow registers (found in NMI 0E:C000-C0D0 and SetPrgBanks 0F:FF4D)
00E8 chrBankShadow 6            ; shadows of MMC3 R0..R5 (CHR banks), written by NMI
00F0 mmc3SelectShadow           ; last value written to $8000 (bank select)
00F2 prgR6Shadow                ; last value written to R6 (PRG bank at $8000)
00F3 prgR7Shadow                ; last value written to R7 (PRG bank at $A000)
00F4 prgBank8000                ; wanted PRG bank in the $8000 window; applied by SetPrgBanks
00F5 prgBankA000                ; wanted PRG bank in the $A000 window (stage-dependent, see stageBankTable)
00F6 bankSwitchBusy             ; INC at entry / DEC at exit of SetPrgBanks
00FA scrollY                    ; second PPUSCROLL write in NMI
00FD nametableSelect            ; bit 0 ORed into PPUCTRL
00FE ppuMaskBuf                 ; value written to PPUMASK by NMI
009C irqVector 2                ; current IRQ handler (JMP (irqVector) from IRQ); each raster split sets the next one

; ---- input / PPU update queues (NMI, bank 0E)
0014 padPressed 2               ; buttons newly pressed this frame, player 1/2 (ReadPads); Up+Down / Left+Right cleared
0016 padHeld 2                  ; raw button state of the previous read, player 1/2
0064 vramQueueFlag              ; non-zero: NMI flushes vramQueue (FlushVramQueue)
0065 vramQueueEnd               ; X index where the vramQueue flush stops; cleared afterwards
00EE renderOffCount             ; >0 while the screen is blanked (ScreenOff/ScreenOn); NMI skips updates
00FC scrollX                    ; copied to irqScrollX each NMI
007C irqScrollY                 ; copy of $F8 for the IRQ splits
007D irqScrollX                 ; copy of scrollX used by IrqSplit1/2
007E irqNametable               ; copy of nametableSelect
0780 vramQueue 128              ; queue: [len, ppuHi, ppuLo, data*len]...  (also vramBuffer: [hi(bit7=end), lo, len-1, data])
